### 第三章	Docker安装

```shell
#1、卸载旧版本Docker
sudo apt remove docker.io docker-compose docker-compose-v2 docker-doc docker-buildx podman-docker containerd runc
#2、添加Docker官方源
sudo apt update
sudo rm -f /etc/apt/keyrings/docker.asc
sudo apt install -y ca-certificates curl
sudo install -m 0755 -d /etc/apt/keyrings
sudo curl -fsSL https://mirrors.aliyun.com/docker-ce/linux/ubuntu/gpg \
    -o /etc/apt/keyrings/docker.asc
sudo chmod a+r /etc/apt/keyrings/docker.asc
sudo tee /etc/apt/sources.list.d/docker.sources <<EOF
Types: deb
URIs: https://mirrors.aliyun.com/docker-ce/linux/ubuntu
Suites: noble
Components: stable
Architectures: amd64
Signed-By: /etc/apt/keyrings/docker.asc
EOF
sudo apt update
#3、安装Docker
sudo apt install -y docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin
#4、查看Docker安装状态
sudo systemctl status docker
sudo docker version
sudo docker compose version
#5、后面如果关闭了可以以下命令进行使能启动
sudo systemctl enable docker
sudo systemctl start docker
#6、取消每次sudo的调用
sudo usermod -aG docker $USER
exit
#7、配置Docker镜像加速
sudo mkdir -p /etc/docker
sudo vim /etc/docker/daemon.json
{
  "registry-mirrors": [
    "https://docker.m.daocloud.io"
  ]
}
sudo systemctl daemon-reload
sudo systemctl restart docker
sudo systemctl status docker
#8、Docker HelloWorld
sudo docker pull hello-world
sudo docker run hello-world
```
